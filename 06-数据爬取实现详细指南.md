## 五、爬虫实现

### 5.1 爬虫基类 (crawlers/base_crawler.py)

```python
import requests
import random
import time
from abc import ABC, abstractmethod
from config import Config
from utils.logger import logger

class BaseCrawler(ABC):
    """爬虫基类"""
    
    def __init__(self):
        self.session = requests.Session()
        self.config = Config()
        
    def get_headers(self):
        """随机获取User-Agent"""
        return {
            'User-Agent': random.choice(self.config.USER_AGENTS),
            'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8',
            'Accept-Language': 'zh-CN,zh;q=0.9,en;q=0.8',
            'Accept-Encoding': 'gzip, deflate',
            'Connection': 'keep-alive'
        }
    
    def request_with_retry(self, url, method='GET', **kwargs):
        """带重试的请求"""
        for i in range(self.config.CRAWLER_MAX_RETRIES):
            try:
                if method == 'GET':
                    response = self.session.get(
                        url, 
                        headers=self.get_headers(),
                        timeout=self.config.CRAWLER_TIMEOUT,
                        **kwargs
                    )
                else:
                    response = self.session.post(
                        url,
                        headers=self.get_headers(),
                        timeout=self.config.CRAWLER_TIMEOUT,
                        **kwargs
                    )
                
                if response.status_code == 200:
                    return response
                    
                logger.warning(f"请求失败，状态码：{response.status_code}，重试第{i+1}次")
                
            except Exception as e:
                logger.error(f"请求异常：{str(e)}，重试第{i+1}次")
                
            time.sleep(self.config.CRAWLER_REQUEST_DELAY * (i + 1))
        
        raise Exception(f"请求失败，已重试{self.config.CRAWLER_MAX_RETRIES}次")
    
    def delay(self):
        """随机延迟"""
        time.sleep(random.uniform(
            self.config.CRAWLER_REQUEST_DELAY,
            self.config.CRAWLER_REQUEST_DELAY * 1.5
        ))
    
    @abstractmethod
    def search_scenic_spot(self, spot_name):
        """搜索景区，获取景区ID"""
        pass
    
    @abstractmethod
    def crawl_reviews(self, spot_id, target_count):
        """爬取评价数据"""
        pass
    
    @abstractmethod
    def crawl_info(self, spot_id):
        """爬取景区基本信息"""
        pass
```

### 5.2 携程爬虫实现 (crawlers/ctrip_crawler.py)

```python
from bs4 import BeautifulSoup
import json
import re
from datetime import datetime
from crawlers.base_crawler import BaseCrawler
from utils.logger import logger

class CtripCrawler(BaseCrawler):
    """携程旅行网爬虫"""
    
    def __init__(self):
        super().__init__()
        self.base_url = "https://you.ctrip.com"
    
    def search_scenic_spot(self, spot_name):
        """
        搜索景区，获取景区ID
        
        注意：由于无法实时访问网站，此处提供的是通用实现逻辑
        实际开发时需要根据携程网站的最新结构调整
        """
        try:
            # 方法1：通过搜索接口
            search_url = f"{self.base_url}/search/all"
            params = {
                'q': spot_name,
                'type': 'sight'
            }
            
            response = self.request_with_retry(search_url, params=params)
            soup = BeautifulSoup(response.text, 'lxml')
            
            # 解析搜索结果，提取景区ID
            # 实际解析逻辑需要根据网站HTML结构编写
            result_items = soup.select('.search-result-item')
            
            for item in result_items:
                title = item.select_one('.title')
                if title and spot_name in title.text:
                    # 从链接中提取景区ID
                    link = item.select_one('a')['href']
                    spot_id = re.search(r'/(\d+)\.html', link)
                    if spot_id:
                        return spot_id.group(1)
            
            # 方法2：如果搜索不到，使用预定义映射
            spot_mapping = {
                '三坊七巷': '109191',
                '鼓山': '53770',
                '平潭岛': '1411471'
            }
            
            return spot_mapping.get(spot_name)
            
        except Exception as e:
            logger.error(f"搜索景区失败：{str(e)}")
            raise
    
    def crawl_reviews(self, spot_id, target_count):
        """
        爬取评价数据
        
        Args:
            spot_id: 景区ID
            target_count: 目标爬取数量
            
        Yields:
            dict: 评价数据
        """
        page = 1
        count = 0
        
        while count < target_count:
            try:
                # 构造评价页面URL
                # 注意：实际URL格式需要根据携程网站调整
                review_url = f"{self.base_url}/sight/{spot_id}/reviews-p{page}.html"
                
                response = self.request_with_retry(review_url)
                soup = BeautifulSoup(response.text, 'lxml')
                
                # 解析评价列表
                review_items = soup.select('.review-item')
                
                if not review_items:
                    logger.info("没有更多评价数据")
                    break
                
                for item in review_items:
                    if count >= target_count:
                        break
                    
                    try:
                        # 提取评价数据
                        review_data = self._parse_review_item(item, spot_id)
                        if review_data:
                            yield review_data
                            count += 1
                    except Exception as e:
                        logger.warning(f"解析单条评价失败：{str(e)}")
                        continue
                
                page += 1
                self.delay()
                
            except Exception as e:
                logger.error(f"爬取第{page}页失败：{str(e)}")
                break
    
    def _parse_review_item(self, item, spot_id):
        """解析单条评价"""
        try:
            # 以下是示例解析逻辑，需要根据实际HTML结构调整
            review_data = {
                'review_id': self._generate_review_id(),
                'spot_id': spot_id,
                'spot_name': '',  # 从页面获取
                'visitor_name': item.select_one('.user-name').text.strip() if item.select_one('.user-name') else '匿名用户',
                'rating': self._extract_rating(item),
                'review_content': item.select_one('.review-content').text.strip() if item.select_one('.review-content') else '',
                'review_images': self._extract_images(item),
                'travel_date': self._extract_date(item, '.travel-date'),
                'review_date': self._extract_date(item, '.review-date'),
                'helpful_count': self._extract_number(item, '.helpful-count'),
                'visitor_level': item.select_one('.user-level').text.strip() if item.select_one('.user-level') else '',
                'visitor_location': item.select_one('.user-location').text.strip() if item.select_one('.user-location') else '',
                'data_source': 'ctrip',
                'crawl_time': datetime.now().strftime('%Y-%m-%d %H:%M:%S')
            }
            
            return review_data
            
        except Exception as e:
            logger.error(f"解析评价项失败：{str(e)}")
            return None
    
    def _extract_rating(self, item):
        """提取评分"""
        try:
            rating_elem = item.select_one('.rating')
            if rating_elem:
                # 可能是 class="star-5" 这样的格式
                rating_class = rating_elem.get('class', [])
                for cls in rating_class:
                    if 'star-' in cls:
                        return float(cls.replace('star-', ''))
            return 5.0  # 默认满分
        except:
            return 5.0
    
    def _extract_images(self, item):
        """提取评价图片"""
        try:
            images = []
            img_elements = item.select('.review-images img')
            for img in img_elements:
                img_url = img.get('src') or img.get('data-src')
                if img_url:
                    images.append(img_url)
            return json.dumps(images, ensure_ascii=False)
        except:
            return '[]'
    
    def _extract_date(self, item, selector):
        """提取日期"""
        try:
            date_elem = item.select_one(selector)
            if date_elem:
                date_text = date_elem.text.strip()
                # 尝试解析日期格式，如"2024-12-10"
                return date_text
            return datetime.now().strftime('%Y-%m-%d')
        except:
            return datetime.now().strftime('%Y-%m-%d')
    
    def _extract_number(self, item, selector):
        """提取数字"""
        try:
            elem = item.select_one(selector)
            if elem:
                text = elem.text.strip()
                numbers = re.findall(r'\d+', text)
                if numbers:
                    return int(numbers[0])
            return 0
        except:
            return 0
    
    def _generate_review_id(self):
        """生成评价ID"""
        import uuid
        return str(uuid.uuid4())
    
    def crawl_info(self, spot_id):
        """爬取景区基本信息"""
        try:
            info_url = f"{self.base_url}/sight/{spot_id}.html"
            response = self.request_with_retry(info_url)
            soup = BeautifulSoup(response.text, 'lxml')
            
            info_data = {
                'spot_id': spot_id,
                'spot_name': soup.select_one('.sight-name').text.strip() if soup.select_one('.sight-name') else '',
                'spot_type': soup.select_one('.sight-type').text.strip() if soup.select_one('.sight-type') else '',
                'location': soup.select_one('.location').text.strip() if soup.select_one('.location') else '',
                'address': soup.select_one('.address').text.strip() if soup.select_one('.address') else '',
                'open_time': soup.select_one('.open-time').text.strip() if soup.select_one('.open-time') else '',
                'ticket_price': self._extract_price(soup),
                'description': soup.select_one('.description').text.strip() if soup.select_one('.description') else '',
                'contact_phone': soup.select_one('.phone').text.strip() if soup.select_one('.phone') else '',
                'rating': self._extract_rating(soup),
                'data_source': 'ctrip',
                'crawl_time': datetime.now().strftime('%Y-%m-%d %H:%M:%S')
            }
            
            return info_data
            
        except Exception as e:
            logger.error(f"爬取景区信息失败：{str(e)}")
            raise
    
    def _extract_price(self, soup):
        """提取门票价格"""
        try:
            price_elem = soup.select_one('.price')
            if price_elem:
                price_text = price_elem.text
                numbers = re.findall(r'\d+\.?\d*', price_text)
                if numbers:
                    return float(numbers[0])
            return 0.0
        except:
            return 0.0
```

### 6.1 HDFS客户端 (utils/hdfs_client.py)

```python
import json
from datetime import datetime
from hdfs import InsecureClient
from config import Config
from utils.logger import logger

class HDFSClient:
    """HDFS客户端封装"""
    
    def __init__(self):
        self.config = Config()
        self.client = InsecureClient(
            f'http://{self.config.HDFS_HOST}:{self.config.HDFS_PORT}',
            user=self.config.HDFS_USER
        )
    
    def save_data(self, data_list, task_id, scenic_spot_name, data_source):
        """
        保存数据到HDFS
        
        Args:
            data_list: 数据列表
            task_id: 任务ID
            scenic_spot_name: 景区名称
            data_source: 数据来源
            
        Returns:
            str: HDFS存储路径
        """
        try:
            # 生成文件路径
            date_str = datetime.now().strftime('%Y%m%d')
            spot_code = self._pinyin_convert(scenic_spot_name)
            filename = f"{data_source}_{spot_code}_{task_id}.json"
            hdfs_path = f"/tourism/raw/{date_str}/{filename}"
            
            # 确保目录存在
            dir_path = f"/tourism/raw/{date_str}"
            if not self.client.status(dir_path, strict=False):
                self.client.makedirs(dir_path)
            
            # 转换为JSON Lines格式
            json_lines = '\n'.join([
                json.dumps(item, ensure_ascii=False) 
                for item in data_list
            ])
            
            # 写入HDFS
            with self.client.write(hdfs_path, encoding='utf-8') as writer:
                writer.write(json_lines)
            
            logger.info(f"数据已保存到HDFS：{hdfs_path}")
            return hdfs_path
            
        except Exception as e:
            logger.error(f"保存数据到HDFS失败：{str(e)}")
            raise
    
    def _pinyin_convert(self, chinese_text):
        """汉字转拼音"""
        from pypinyin import lazy_pinyin
        pinyin_list = lazy_pinyin(chinese_text)
        return ''.join(pinyin_list)
```

```python
                'visitor_name': DataCleaner.clean_text(raw_data.get('visitor_name', '')),
                'rating': DataCleaner.validate_rating(raw_data.get('rating')),
                'review_content': DataCleaner.clean_text(raw_data.get('review_content', '')),
                'review_images': raw_data.get('review_images', '[]'),
                'travel_date': DataCleaner.parse_date(raw_data.get('travel_date')),
                'review_date': DataCleaner.parse_date(raw_data.get('review_date')),
                'helpful_count': int(raw_data.get('helpful_count', 0)),
                'visitor_level': raw_data.get('visitor_level', ''),
                'visitor_location': DataCleaner.clean_text(raw_data.get('visitor_location', '')),
                'data_source': raw_data.get('data_source', ''),
                'crawl_time': raw_data.get('crawl_time', datetime.now().strftime('%Y-%m-%d %H:%M:%S'))
            }
            
            return cleaned_data
            
        except Exception as e:
            logger.error(f"清洗数据失败：{str(e)}")
            return None
    
    @staticmethod
    def clean_text(text):
        """清洗文本"""
        if not text:
            return ''
        
        # 去除多余空格
        text = re.sub(r'\s+', ' ', text)
        
        # 去除特殊字符
        text = re.sub(r'[^\w\s\u4e00-\u9fff，。！？、；：""''《》（）【】]', '', text)
        
        return text.strip()
    
    @staticmethod
    def validate_rating(rating):
        """验证评分"""
        try:
            rating = float(rating)
            if 1 <= rating <= 5:
                return rating
            return 5.0
        except:
            return 5.0
    
    @staticmethod
    def parse_date(date_str):
        """解析日期"""
        if not date_str:
            return datetime.now().strftime('%Y-%m-%d')
        
        try:
            # 尝试多种日期格式
            formats = ['%Y-%m-%d', '%Y/%m/%d', '%Y年%m月%d日']
            for fmt in formats:
                try:
                    dt = datetime.strptime(date_str, fmt)
                    return dt.strftime('%Y-%m-%d')
                except:
                    continue
            
            return datetime.now().strftime('%Y-%m-%d')
        except:
            return datetime.now().strftime('%Y-%m-%d')
```

### 7.2 爬取任务 (tasks/crawl_task.py)

```python
from datetime import datetime
from celery_worker import celery_app
from crawlers.ctrip_crawler import CtripCrawler
from crawlers.qunar_crawler import QunarCrawler
from crawlers.meituan_crawler import MeituanCrawler
from utils.hdfs_client import HDFSClient
from utils.data_cleaner import DataCleaner
from utils.logger import logger
from models.database import SessionLocal
from models.task import CrawlerTask

@celery_app.task(bind=True)
def execute_crawl_task(self, task_id, scenic_spot_name, data_source, target_count, data_type):
    """
    执行爬取任务
    
    Args:
        task_id: 任务ID
        scenic_spot_name: 景区名称
        data_source: 数据来源
        target_count: 目标数量
        data_type: 数据类型
    """
    db = SessionLocal()
    
    try:
        # 更新任务状态为运行中
        task = db.query(CrawlerTask).filter(CrawlerTask.id == task_id).first()
        task.status = 'running'
        task.start_time = datetime.now()
        task.progress = 0
        db.commit()
        
        # 选择爬虫
        crawler = get_crawler(data_source)
        
        # 搜索景区ID
        spot_id = crawler.search_scenic_spot(scenic_spot_name)
        if not spot_id:
            raise Exception(f"未找到景区：{scenic_spot_name}")
        
        # 爬取数据
        data_list = []
        
        if data_type == 'review':
            for i, raw_data in enumerate(crawler.crawl_reviews(spot_id, target_count)):
                # 清洗数据
                cleaned_data = DataCleaner.clean_review_data(raw_data)
                if cleaned_data:
                    data_list.append(cleaned_data)
                
                # 更新进度
                progress = int((i + 1) / target_count * 100)
                task.progress = progress
                task.actual_count = i + 1
                db.commit()
                
                logger.info(f"任务{task_id}进度：{progress}%")
        
        elif data_type == 'info':
            info_data = crawler.crawl_info(spot_id)
            if info_data:
                data_list.append(info_data)
                task.actual_count = 1
                task.progress = 100
                db.commit()
        
        # 保存到HDFS
        hdfs_client = HDFSClient()
        hdfs_path = hdfs_client.save_data(data_list, task_id, scenic_spot_name, data_source)
        
        # 更新任务状态为成功
        task.status = 'success'
        task.progress = 100
        task.end_time = datetime.now()
        task.hdfs_path = hdfs_path
        db.commit()
        
        logger.info(f"任务{task_id}执行成功")
        
    except Exception as e:
        # 更新任务状态为失败
        logger.error(f"任务{task_id}执行失败：{str(e)}")
        task = db.query(CrawlerTask).filter(CrawlerTask.id == task_id).first()
        task.status = 'failed'
        task.error_msg = str(e)
        task.end_time = datetime.now()
        db.commit()
        raise
    
    finally:
        db.close()

def get_crawler(data_source):
    """获取爬虫实例"""
    crawlers = {
        'ctrip': CtripCrawler,
        'qunar': QunarCrawler,
        'meituan': MeituanCrawler
    }
    
    crawler_class = crawlers.get(data_source)
    if not crawler_class:
        raise Exception(f"不支持的数据源：{data_source}")
    
    return crawler_class()
```

```python
from flask import Flask, request, jsonify

@app.route('/api/crawler/start', methods=['POST'])
def start_crawl():
    """启动爬取任务"""
    try:
        data = request.get_json()
        
        # 验证参数
        required_fields = ['task_name', 'scenic_spot_name', 'data_source', 'target_count', 'data_type']
        for field in required_fields:
            if field not in data:
                return jsonify({'code': 400, 'msg': f'缺少参数：{field}'}), 400
        
        db = SessionLocal()
        
        # 创建任务记录
        task = CrawlerTask(
            task_name=data['task_name'],
            scenic_spot_name=data['scenic_spot_name'],
            data_source=data['data_source'],
            target_count=data['target_count'],
            status='pending',
            create_by=1  # TODO: 从token中获取用户ID
        )
        
        db.add(task)
        db.commit()
        db.refresh(task)
        
        task_id = task.id
        
        # 异步执行爬取任务
        celery_task = execute_crawl_task.delay(
            task_id,
            data['scenic_spot_name'],
            data['data_source'],
            data['target_count'],
            data['data_type']
        )
        
        db.close()
        
        return jsonify({
            'code': 200,
            'msg': '任务已启动',
            'data': {
                'task_id': task_id,
                'celery_task_id': celery_task.id,
                'status': 'running',
                'create_time': datetime.now().strftime('%Y-%m-%d %H:%M:%S')
            }
        })
        
    except Exception as e:
        logger.error(f"启动任务失败：{str(e)}")
```
