```java
package com.tourism.tourismspringboot.config;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.jdbc.datasource.DriverManagerDataSource;

import javax.sql.DataSource;

/**
 * Hive数据源配置
 */
@Configuration
public class HiveConfig {

    @Value("${hive.jdbc.url}")
    private String hiveUrl;

    @Value("${hive.jdbc.username}")
    private String username;

    @Value("${hive.jdbc.password}")
    private String password;

    @Value("${hive.jdbc.driver-class-name}")
    private String driverClassName;

    @Bean(name = "hiveDataSource")
    public DataSource hiveDataSource() {
        DriverManagerDataSource dataSource = new DriverManagerDataSource();
        dataSource.setDriverClassName(driverClassName);
        dataSource.setUrl(hiveUrl);
        dataSource.setUsername(username);
        dataSource.setPassword(password);
        return dataSource;
    }

    @Bean(name = "hiveJdbcTemplate")
    public JdbcTemplate hiveJdbcTemplate() {
        return new JdbcTemplate(hiveDataSource());
    }
}
```

```java
package com.tourism.tourismspringboot.utils;

import io.jsonwebtoken.Claims;
import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.SignatureAlgorithm;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

import java.util.Date;
import java.util.HashMap;
import java.util.Map;

@Component
public class JwtUtil {

    @Value("${jwt.secret}")
    private String secret;

    @Value("${jwt.expiration}")
    private Long expiration;

    /**
     * 生成Token
     */
    public String generateToken(Long userId, String username) {
        Map<String, Object> claims = new HashMap<>();
        claims.put("userId", userId);
        claims.put("username", username);

        return Jwts.builder()
                .setClaims(claims)
                .setSubject(username)
                .setIssuedAt(new Date())
                .setExpiration(new Date(System.currentTimeMillis() + expiration * 1000))
                .signWith(SignatureAlgorithm.HS512, secret)
                .compact();
    }

    /**
     * 从Token中获取用户ID
     */
    public Long getUserIdFromToken(String token) {
        Claims claims = getClaimsFromToken(token);
        return claims != null ? Long.valueOf(claims.get("userId").toString()) : null;
    }

    /**
     * 从Token中获取用户名
     */
    public String getUsernameFromToken(String token) {
        Claims claims = getClaimsFromToken(token);
        return claims != null ? claims.getSubject() : null;
    }

    /**
     * 验证Token是否有效
     */
    public boolean validateToken(String token) {
        try {
            Claims claims = getClaimsFromToken(token);
            return claims != null && !isTokenExpired(claims);
        } catch (Exception e) {
            return false;
        }
    }

    /**
     * 获取Claims
     */
    private Claims getClaimsFromToken(String token) {
        try {
            return Jwts.parser()
                    .setSigningKey(secret)
                    .parseClaimsJws(token)
                    .getBody();
        } catch (Exception e) {
            return null;
        }
    }

    /**
     * 判断Token是否过期
     */
    private boolean isTokenExpired(Claims claims) {
        Date expiration = claims.getExpiration();
        return expiration.before(new Date());
    }
}
```

#### 3. ScenicAnalysisServiceImpl.java - **Hive查询实现**
```java
package com.tourism.tourismspringboot.service.impl;

import com.tourism.tourismspringboot.service.IScenicAnalysisService;
import com.tourism.tourismspringboot.vo.*;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.beans.factory.annotation.Qualifier;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Service;

import java.util.*;

@Service
public class ScenicAnalysisServiceImpl implements IScenicAnalysisService {

    @Autowired
    @Qualifier("hiveJdbcTemplate")
    private JdbcTemplate hiveJdbcTemplate;

    @Override
    public List<ScenicVO> getScenicList() {
        String sql = 
            "SELECT " +
            "    spot_id, " +
            "    spot_name, " +
            "    COUNT(*) as total_reviews, " +
            "    AVG(rating) as avg_rating " +
            "FROM fuzhou_reviews " +
            "GROUP BY spot_id, spot_name " +
            "ORDER BY total_reviews DESC";

        return hiveJdbcTemplate.query(sql, (rs, rowNum) -> {
            ScenicVO vo = new ScenicVO();
            vo.setSpotId(rs.getString("spot_id"));
            vo.setSpotName(rs.getString("spot_name"));
            vo.setTotalReviews(rs.getLong("total_reviews"));
            vo.setAvgRating(rs.getDouble("avg_rating"));
            return vo;
        });
    }

    @Override
    public List<VisitorCountVO> getVisitorCount(String spotId, String startDate, String endDate) {
        String sql = 
            "SELECT " +
            "    travel_date as stat_date, " +
            "    COUNT(*) as visitor_count " +
            "FROM fuzhou_reviews " +
            "WHERE spot_id = ? " +
            "  AND travel_date >= ? " +
            "  AND travel_date <= ? " +
            "GROUP BY travel_date " +
            "ORDER BY travel_date";

        return hiveJdbcTemplate.query(sql, 
            new Object[]{spotId, startDate, endDate},
            (rs, rowNum) -> {
                VisitorCountVO vo = new VisitorCountVO();
                vo.setStatDate(rs.getString("stat_date"));
                vo.setVisitorCount(rs.getLong("visitor_count"));
                return vo;
            });
    }

    @Override
    public ReviewStatsVO getReviewStats(String spotId) {
        // 查询总体统计
        String sql = 
            "SELECT " +
            "    spot_name, " +
            "    COUNT(*) as total_reviews, " +
            "    AVG(rating) as avg_rating " +
            "FROM fuzhou_reviews " +
            "WHERE spot_id = ? " +
            "GROUP BY spot_name";

        ReviewStatsVO stats = hiveJdbcTemplate.queryForObject(sql, 
            new Object[]{spotId},
            (rs, rowNum) -> {
                ReviewStatsVO vo = new ReviewStatsVO();
                vo.setSpotName(rs.getString("spot_name"));
                vo.setTotalReviews(rs.getLong("total_reviews"));
                vo.setAvgRating(rs.getDouble("avg_rating"));
                return vo;
            });

        // 查询评分分布
        String distSql = 
            "SELECT " +
            "    CAST(rating AS INT) as rating_level, " +
            "    COUNT(*) as count " +
            "FROM fuzhou_reviews " +
            "WHERE spot_id = ? " +
            "GROUP BY CAST(rating AS INT)";

        Map<String, Long> distribution = new HashMap<>();
        hiveJdbcTemplate.query(distSql, new Object[]{spotId}, (rs) -> {
            distribution.put(rs.getString("rating_level"), rs.getLong("count"));
        });

        stats.setRatingDistribution(distribution);

        // 计算好评率（4分及以上）
        Long positive = distribution.getOrDefault("4", 0L) + distribution.getOrDefault("5", 0L);
        stats.setPositiveRate(positive * 100.0 / stats.getTotalReviews());

        return stats;
    }

    @Override
    public SeasonPatternVO getSeasonPattern(String spotId, Integer year) {
        String sql = 
            "SELECT " +
            "    MONTH(travel_date) as month, " +
            "    COUNT(*) as visitor_count " +
            "FROM fuzhou_reviews " +
            "WHERE spot_id = ? " +
            "  AND YEAR(travel_date) = ? " +
            "GROUP BY MONTH(travel_date) " +
            "ORDER BY month";

        List<MonthlyStatVO> monthlyStats = hiveJdbcTemplate.query(sql,
            new Object[]{spotId, year},
            (rs, rowNum) -> {
                MonthlyStatVO vo = new MonthlyStatVO();
                vo.setMonth(rs.getString("month"));
                vo.setVisitorCount(rs.getLong("visitor_count"));
                // 判断季节
                int month = rs.getInt("month");
                if (month >= 3 && month <= 5) {
                    vo.setSeasonType("春季");
                } else if (month >= 6 && month <= 8) {
                    vo.setSeasonType("夏季");
                } else if (month >= 9 && month <= 11) {
                    vo.setSeasonType("秋季");
                } else {
                    vo.setSeasonType("冬季");
                }
                return vo;
            });

        SeasonPatternVO pattern = new SeasonPatternVO();
        pattern.setYear(year);
        pattern.setMonthlyStats(monthlyStats);

        return pattern;
    }

    @Override
    public Map<String, Object> getOverview() {
        Map<String, Object> overview = new HashMap<>();

        // 总景区数
        String scenicSql = "SELECT COUNT(DISTINCT spot_id) as count FROM fuzhou_reviews";
        overview.put("totalScenicSpots", hiveJdbcTemplate.queryForObject(scenicSql, Long.class));

        // 总评论数
        String reviewSql = "SELECT COUNT(*) as count FROM fuzhou_reviews";
        overview.put("totalReviews", hiveJdbcTemplate.queryForObject(reviewSql, Long.class));

        // 平均评分
        String ratingSql = "SELECT AVG(rating) as avg FROM fuzhou_reviews";
        overview.put("avgRating", hiveJdbcTemplate.queryForObject(ratingSql, Double.class));

        return overview;
    }
}
```

## 核心技术点总结

1. **Hive JDBC连接**: 通过HiveConfig配置独立的Hive数据源
2. **数据查询**: 使用JdbcTemplate执行Hive SQL查询
3. **数据格式**: Hive表使用JsonSerDe直接解析Flask爬取的JSONL数据
4. **跨域配置**: CorsConfig支持前端调用
5. **接口文档**: Knife4j自动生成API文档
6. **统一响应**: Result统一封装响应格式

这套代码实现了从Flask爬取数据→存储到Hadoop→Hive分析→SpringBoot查询→前端展示的完整数据流！
